export function ringOffset(base: Uint8Array, delta: number): Uint8Array {
	const out = new Uint8Array(base.length)
	let carry = delta
	for (let i = base.length - 1; i >= 0; i--) {
		const v = (base[i] ?? 0) + carry

}
