```ts
// Handler sends half a framed reply then close(); dialer (never writes) iterates RAW stream.
// Question: does the raw async iterator ever end/throw after the remote half-close?
import { createMemoryNode } from './helpers/libp2p.js'
import { openRpcStream } from '../src/rpc/protocols.js'

const PROTOCOL = '/optimystic/default/fret/1.0.0/ping'
const enc = new TextEncoder()
const body = enc.encode(JSON.stringify({ ok: true, ts: 1, size_estimate: 42, confidence: 0.5 }))
const half = Math.floor(body.length / 2)
const partial = new Uint8Array(1 + half)      // varint prefix for len < 128 is one byte
partial[0] = body.length
partial.set(body.subarray(0, half), 1)

const t0 = Date.now()
const ts = () => `[+${Date.now() - t0}ms]`
const a = await createMemoryNode(); await a.start()
const b = await createMemoryNode(); await b.start()

await a.handle(PROTOCOL, async (stream: any) => {
	console.log(ts(), 'handler: sending', partial.length, 'bytes then close()')
	stream.send(partial)
	await stream.close()
	console.log(ts(), 'handler: close() resolved; status', stream.status, 'write', stream.writeStatus, 'read', stream.readStatus)
})

await b.dial(a.getMultiaddrs()[0]!)
const stream: any = await openRpcStream(b as any, a.peerId, [PROTOCOL])
console.log(ts(), 'dialer stream open; status', stream.status, 'write', stream.writeStatus, 'read', stream.readStatus)
for (const ev of ['message', 'close', 'remoteCloseWrite', 'remoteCloseRead']) {
	try { stream.addEventListener(ev, () => console.log(ts(), `event: ${ev}; status`, stream.status, 'write', stream.writeStatus, 'read', stream.readStatus)) } catch {}
}
const watchdog = setTimeout(() => {
	console.log(ts(), 'WATCHDOG: iterator still pending after 8s; status', stream.status, 'write', stream.writeStatus, 'read', stream.readStatus)
	process.exit(2)
}, 8000)
try {
	for await (const chunk of stream) {
		console.log(ts(), 'chunk:', chunk.byteLength ?? chunk.length, 'bytes')
	}
	console.log(ts(), 'iterator ENDED cleanly')
} catch (err) {
	console.log(ts(), 'iterator THREW:', (err as Error).name, '-', (err as Error).message)
}
clearTimeout(watchdog)
await b.stop(); await a.stop()
console.log(ts(), 'done')
process.exit(0)
```
